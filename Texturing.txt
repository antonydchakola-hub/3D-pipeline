function handleTexturingWorkflow(e) {
  var range = e.range;
  var texSheet = range.getSheet();
  var row = range.getRow();
  var col = range.getColumn();
  
  var statusCol = 10; // Column J

  if (col !== statusCol || row < 2) return;
  
  var statusValue = range.getValue() ? range.getValue().toString().trim() : "";
  if (statusValue === "Sent to Lighting" || statusValue === "Sent to Modelling" || statusValue === "Archived Rework" || statusValue === "Archived Split") return;
  
  if (statusValue !== "Done" && statusValue !== "Rework (Modelling)" && statusValue !== "Rework (Texturing)" && statusValue !== "Split" && statusValue !== "Split Done") return;
  
  var tokenCol = 16; // Column P (Lock to prevent duplicates)
  if (texSheet.getRange(row, tokenCol).getValue() === "PROCESSED") return; 

  var LIGHTING_SHEET_ID = "14RSsDESwBiIsjlF5dqoYZmkExSY6uFoTHYuYbmFSK-g";
  var MODELLING_SHEET_ID = "1Bz7S3lcnA3NaqT9MvuhNjY79PBrHEq-FMjJndfBP_2U";
  var MANAGER_SHEET_ID = "14N_O6oR9qb12w77TJLv0EEOb3_b-siKdzgEPjtaMKmk";
  var ARTIST_SHEET_ID = "1SaLO_U_9NbXE5KarsG9XSOeCWkKKr1LQhLeZRMoZNIQ";

  // Helper Function: Safe Tab Grabber
  var getManagerTab = function(managerId, activeTabName) {
    var ss = SpreadsheetApp.openById(managerId);
    var sheet = ss.getSheetByName(activeTabName);
    if (!sheet) {
      sheet = ss.getSheets()[0]; 
    }
    return sheet;
  };

  // --- MANAGER SYNC FUNCTION ---
  var syncToManagerSheet = function(targetTcn, artistName, textStatus, modStatus, incrementField) {
    if (!targetTcn) return;
    try {
      var cleanTarget = targetTcn.toString().split('.')[0].trim();
      if (cleanTarget === "") return;

      var mgrSheet = getManagerTab(MANAGER_SHEET_ID, texSheet.getName());
      var mgrData = mgrSheet.getDataRange().getValues();
      
      for (var i = 0; i < mgrData.length; i++) {
        for (var c = 0; c <= 3; c++) { 
          if (!mgrData[i][c]) continue;
          var checkStr = mgrData[i][c].toString().split('.')[0].trim();
          
          if (checkStr !== "" && checkStr === cleanTarget) {
            var mgrRow = i + 1;
            
            if (artistName !== null) mgrSheet.getRange(mgrRow, 10).setValue(artistName); // Text Artist (Col J)
            if (textStatus !== null) mgrSheet.getRange(mgrRow, 11).setValue(textStatus); // Text Status (Col K)
            if (modStatus !== null) mgrSheet.getRange(mgrRow, 9).setValue(modStatus); // Mod Status (Col I)
            
            if (incrementField === "mod") {
              var modReworkCell = mgrSheet.getRange(mgrRow, 16); // Mod Reworks (Col P)
              var currentCount = modReworkCell.getValue();
              var cleanCount = (typeof currentCount === 'number' && !isNaN(currentCount)) ? currentCount : 0;
              modReworkCell.setValue(cleanCount + 1);
            } 
            else if (incrementField === "text") {
              var textReworkCell = mgrSheet.getRange(mgrRow, 17); // Text Reworks (Col Q)
              var currentCount = textReworkCell.getValue();
              var cleanCount = (typeof currentCount === 'number' && !isNaN(currentCount)) ? currentCount : 0;
              textReworkCell.setValue(cleanCount + 1);
            }
            
            return true;
          }
        }
      }
    } catch (err) {
      console.log("Manager Sync Error from Texturing: " + err.message);
    }
    return false;
  };

  // --- ARTIST SHEET SYNC FUNCTION ---
  var syncToArtistSheet = function(artistName, type, complexity, allotTime, timeSpent, reworkTime) {
    if (!artistName || artistName.toString().trim() === "") return;
    try {
      var artistSs = SpreadsheetApp.openById(ARTIST_SHEET_ID);
      var artistSheet = artistSs.getSheetByName("Texturing");
      if (!artistSheet) {
        artistSheet = artistSs.getSheets()[0];
      }
      
      var data = artistSheet.getDataRange().getValues();
      var artistRow = -1;
      var cleanSearchName = artistName.toString().trim().toLowerCase();
      
      for (var i = 1; i < data.length; i++) {
        if (data[i][0] && data[i][0].toString().trim().toLowerCase() === cleanSearchName) {
          artistRow = i + 1;
          break;
        }
      }
      
      if (artistRow === -1) {
        console.log("Artist '" + artistName + "' not found in Artist Sheet.");
        return;
      }
      
      var complexityCols = {
        "very simple": 3,
        "simple": 4,
        "medium": 5,
        "hard": 6,
        "very hard": 7
      };
      
      var valAllot = Number(allotTime) || 0;
      var valSpent = Number(timeSpent) || 0;
      var valRework = Number(reworkTime) || 0;
      
      if (type.toString().trim().toLowerCase() === "new") {
        var cellModels = artistSheet.getRange(artistRow, 2);
        cellModels.setValue((Number(cellModels.getValue()) || 0) + 1);
        
        var cleanComp = complexity.toString().trim().toLowerCase();
        if (complexityCols[cleanComp]) {
          var cellComp = artistSheet.getRange(artistRow, complexityCols[cleanComp]);
          cellComp.setValue((Number(cellComp.getValue()) || 0) + 1);
        }
      }
      
      var cellHours = artistSheet.getRange(artistRow, 10);
      cellHours.setValue((Number(cellHours.getValue()) || 0) + valAllot);
      
      var cellTimeSpent = artistSheet.getRange(artistRow, 11);
      cellTimeSpent.setValue((Number(cellTimeSpent.getValue()) || 0) + valSpent + valRework);
      
      var cellReworkTime = artistSheet.getRange(artistRow, 12);
      cellReworkTime.setValue((Number(cellReworkTime.getValue()) || 0) + valRework);
      
    } catch (err) {
      console.log("Artist Sheet Sync Error from Texturing: " + err.message);
    }
  };

  // --- RUN WORKFLOW EXECUTIONS ---
  try {
    texSheet.getRange(row, tokenCol).setValue("PROCESSED");
    SpreadsheetApp.flush(); 

    // SWAPPED: TCN is now source 2, Type is now source 4
    var MAP = {
      "No": {src: 1, dest: 1}, 
      "TCN": {src: 2, dest: 2}, 
      "Comments": {src: 3, dest: 3},
      "Type": {src: 4, dest: 4}, 
      "Priority": {src: 5, dest: 5}, 
      "Complexity": {src: 6, dest: 6}, 
      "Allot": {src: 7, dest: 7}
    };
    
    var checkTcnExists = function(destSheet, tcn) {
      var data = destSheet.getDataRange().getValues();
      return data.some(function(r) { 
        if (!r[1] || !tcn) return false; // r[1] corresponds to TCN (Col B)
        return r[1].toString().split('.')[0].trim() === tcn.toString().split('.')[0].trim();
      });
    };

    var currentTcn = texSheet.getRange(row, 2).getValue(); // Col B (2) is TCN
    var currentArtist = texSheet.getRange(row, 9).getValue(); // Col I (9) is Text Artist

    var rowType = texSheet.getRange(row, 4).getValue();       // Col D (4) is Type
    var rowComplexity = texSheet.getRange(row, 6).getValue(); // Col F (6) is Complexity
    var rowAllotTime = texSheet.getRange(row, 11).getValue(); // Col K (11) is Allot
    var rowTimeSpent = texSheet.getRange(row, 12).getValue(); // Col L (12) is Time Spent
    var rowReworkTime = texSheet.getRange(row, 13).getValue(); // Col M (13) is Rework Time

    if (statusValue === "Done") {
      syncToArtistSheet(currentArtist, rowType, rowComplexity, rowAllotTime, rowTimeSpent, rowReworkTime);

      var lightSheet = SpreadsheetApp.openById(LIGHTING_SHEET_ID).getSheetByName(texSheet.getName());
      if (!lightSheet) throw new Error("Tab name mismatch in Lighting Sheet!");
      
      var newRow = [];
      for (var key in MAP) { newRow[MAP[key].dest - 1] = texSheet.getRange(row, MAP[key].src).getValue(); }
      
      // newRow[3] is Type (index 3), newRow[1] is TCN (index 1)
      newRow[3] = checkTcnExists(lightSheet, newRow[1]) ? "rework" : "new"; 
      
      lightSheet.appendRow(newRow);
      texSheet.getRange(row, statusCol).setValue("Sent to Lighting");
      
      syncToManagerSheet(currentTcn, currentArtist, "Done", null, null);
    } 
    else if (statusValue === "Split") {
      var lastRowInTex = texSheet.getLastRow();
      var splitRow = [];
      
      for (var key in MAP) { splitRow[MAP[key].dest - 1] = texSheet.getRange(row, MAP[key].src).getValue(); }
      
      splitRow[3] = "new"; // Type is index 3
      splitRow[1] = (splitRow[1] ? splitRow[1].toString() : "") + "-"; // TCN is index 1
      
      texSheet.getRange(lastRowInTex + 1, 1, 1, splitRow.length).setValues([splitRow]);
      texSheet.getRange(lastRowInTex + 1, statusCol).setValue(""); 
      texSheet.getRange(row, statusCol).setValue("Archived Split");
    }
    else if (statusValue === "Split Done") {
      syncToArtistSheet(currentArtist, rowType, rowComplexity, rowAllotTime, rowTimeSpent, rowReworkTime);
    }
    else {
      var ui = SpreadsheetApp.getUi();
      var reasonResponse = ui.prompt('Rework Reason', 'Please enter the feedback description comment:', ui.ButtonSet.OK_CANCEL);
      
      if (reasonResponse.getSelectedButton() !== ui.Button.OK) {
        texSheet.getRange(row, statusCol).setValue("");
        texSheet.getRange(row, tokenCol).setValue("");
        return;
      }
      var comment = reasonResponse.getResponseText();

      if (statusValue === "Rework (Modelling)") {
        syncToArtistSheet(currentArtist, rowType, rowComplexity, rowAllotTime, rowTimeSpent, rowReworkTime);

        var modSheet = SpreadsheetApp.openById(MODELLING_SHEET_ID).getSheetByName(texSheet.getName());
        if (!modSheet) throw new Error("Tab name mismatch in Modeling Sheet!");
        
        var modData = modSheet.getDataRange().getValues();
        var foundInMod = false;
        
        for (var i = modData.length - 1; i >= 1; i--) {
          if (modData[i][1] == currentTcn) { // Col B is TCN
            var modRow = i + 1;
            var duplicatedRowValues = [];
            for (var key in MAP) {
              duplicatedRowValues[MAP[key].dest - 1] = modSheet.getRange(modRow, MAP[key].src).getValue();
            }
            
            duplicatedRowValues[3] = "rework"; // Type
            duplicatedRowValues[2] = comment;  // Comments
            
            var lastRowInMod = modSheet.getLastRow();
            modSheet.getRange(lastRowInMod + 1, 1, 1, duplicatedRowValues.length).setValues([duplicatedRowValues]);
            modSheet.getRange(lastRowInMod + 1, statusCol).setValue(""); 
            modSheet.getRange(modRow, statusCol).setValue("Archived Rework");
            foundInMod = true;
            break; 
          }
        }
        
        if (!foundInMod) throw new Error("No active row found for TCN " + currentTcn + " in Modeling!");
        
        texSheet.getRange(row, statusCol).setValue("Sent to Modelling");
        syncToManagerSheet(currentTcn, null, "", "Rework", "mod"); 
      } 
      else if (statusValue === "Rework (Texturing)") {
        syncToArtistSheet(currentArtist, rowType, rowComplexity, rowAllotTime, rowTimeSpent, rowReworkTime);

        var lastRowInTex2 = texSheet.getLastRow();
        var texRowValues = [];
        
        for (var key in MAP) {
          texRowValues[MAP[key].dest - 1] = texSheet.getRange(row, MAP[key].src).getValue();
        }
        
        texRowValues[3] = "rework"; // Type
        texRowValues[2] = comment;  // Comments
        
        texSheet.getRange(lastRowInTex2 + 1, 1, 1, texRowValues.length).setValues([texRowValues]);
        texSheet.getRange(lastRowInTex2 + 1, statusCol).setValue(""); 
        
        if (currentArtist) {
          texSheet.getRange(lastRowInTex2 + 1, 9).setValue(currentArtist); 
        }
        
        texSheet.getRange(row, statusCol).setValue("Archived Rework");
        syncToManagerSheet(currentTcn, null, "Rework", null, "text"); 
      }
    }
    
    texSheet.getRange(row, tokenCol).setValue("");
    
  } catch (err) {
    texSheet.getRange(row, tokenCol).setValue("ERROR: " + err.message);
  }
}
